# --- Core PDF Processing ---
pymupdf==1.24.9              # Tier 1: Native digital PDF extraction (fast path)
Pillow==10.4.0               # Image processing for scanned page detection

# --- Layout-Aware OCR (Tier 2) ---
# Docling: IBM Research layout-aware document converter (matches cloud accuracy)
# NOTE: the 2.5.0 pin below was never actually installed/verified in this repo
# until 2026-09-28 — it turned out to need a newer pydantic than the 2.8.2
# pinned below supports (docling's pydantic model definitions raised
# `AttributeError: force_full_page_ocr` at import time on 2.8.2). Pinning to
# the versions actually verified working here instead of guessing at 2.5.0's
# compatibility with today's other pins.
docling==2.130.0
docling-core==2.99.0

# --- NLP & Entity Extraction (Tier 3) ---
# Tier 3 (pipeline/tier3_entity_extractor.py) is pure regex/rules-based and
# does NOT use spaCy — nothing imports it on the extraction path. spaCy is
# listed here only because api/server.py's health check reports its presence
# as an informational signal for a possible future NER upgrade. Uninstalling
# it does not affect extraction accuracy or any current feature.
spacy==3.7.6

# --- FastAPI Server ---
fastapi==0.115.0
uvicorn[standard]==0.30.6
python-multipart==0.0.9      # For file upload endpoints

# --- Data Models & Validation ---
# Must stay >=2.9 for docling==2.130.0 compatibility (see note above) — the
# rest of this codebase's Pydantic usage (BaseModel, Field, ConfigDict) is
# unaffected either way.
pydantic==2.13.5
pydantic-settings==2.15.0

# --- Background Job Queue ---
# FastAPI BackgroundTasks running in a thread pool, with job status persisted
# to a local SQLite file (api/job_store.py) so job history survives restarts.
# This is a single-instance design: fine for the air-gapped single-node
# deployment this system targets. If this ever needs multiple API instances
# behind a load balancer, swap JOB_STORE for Redis/Postgres-backed Celery.
